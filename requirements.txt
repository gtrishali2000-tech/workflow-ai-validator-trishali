Workflows AI – Live Coding Round Repo Skeleton (Node + TypeScript)

This canvas contains a minimal, runnable skeleton you can hand to candidates. It matches our finalized assignment and includes:
	•	Types & JSON Schema (AJV-ready)
	•	Light RAG over local recipes (simple TF–IDF retriever)
	•	LLM adapter (OpenAI by default, easily swappable)
	•	Validation layer (design left open-ended; baseline AJV provided)
	•	CLI (run.ts) and Eval harness (eval.ts)

Candidate owns prompting, retrieval tuning, validation+repair strategy, and eval improvements. This skeleton only wires basics.

src/llm.ts (adapter; candidates may modify freely)

import OpenAI from "openai";

export interface LLMResponse {
  raw: string;
}

export async function callLLM(prompt: string, model = process.env.OPENAI_MODEL || "gpt-4.1"): Promise<LLMResponse> {
  const apiKey = process.env.OPENAI_API_KEY;
  if (!apiKey) throw new Error("OPENAI_API_KEY missing");
  const client = new OpenAI({ apiKey });

  // Minimal text call. Candidates can swap to JSON mode / tools.
  const res = await client.chat.completions.create({
    model,
    messages: [
      { role: "system", content: "You are a workflow builder that returns ONLY JSON per the provided schema. No prose." },
      { role: "user", content: prompt }
    ],
    temperature: 0.2
  });
  const raw = res.choices[0]?.message?.content ?? "";
  return { raw };
}

src/retriever.ts (tiny TF–IDF over recipe prompts)

import fs from "fs";
import path from "path";

export interface Recipe { prompt: string; workflow: any; }

function tokenize(s: string) {
  return s.toLowerCase().replace(/[^a-z0-9]+/g, " ").trim().split(/\s+/).filter(Boolean);
}

export class Retriever {
  private recipes: Recipe[] = [];
  private docFreq: Map<string, number> = new Map();
  private docsTokens: string[][] = [];

  constructor(file = path.resolve("data/recipes.json")) {
    const data = JSON.parse(fs.readFileSync(file, "utf-8")) as Recipe[];
    this.recipes = data;
    this.docsTokens = data.map(r => tokenize(r.prompt));
    const df = new Map<string, number>();
    this.docsTokens.forEach(tokens => {
      const uniq = new Set(tokens);
      uniq.forEach(t => df.set(t, (df.get(t) || 0) + 1));
    });
    this.docFreq = df;
  }

  retrieve(query: string, k = 3): Recipe[] {
    const q = tokenize(query);
    const N = this.recipes.length;
    const scores = this.docsTokens.map((doc, i) => {
      const tf = new Map<string, number>();
      doc.forEach(t => tf.set(t, (tf.get(t) || 0) + 1));
      let score = 0;
      q.forEach(t => {
        const df = this.docFreq.get(t) || 0;
        if (df === 0) return;
        const idf = Math.log((N + 1) / (df + 1));
        const freq = tf.get(t) || 0;
        score += freq * idf;
      });
      return { i, score };
    });
    scores.sort((a, b) => b.score - a.score);
    return scores.slice(0, k).map(s => this.recipes[s.i]);
  }
}

src/validation.ts (baseline AJV; candidate may replace/extend)

import Ajv from "ajv";
import addFormats from "ajv-formats";
import schema from "../schema/workflow.schema.json" assert { type: "json" };

const ajv = new Ajv({ allErrors: true, strict: false });
addFormats(ajv);
const validate = ajv.compile(schema);

export function validateWorkflow(obj: unknown): { valid: boolean; errors?: string } {
  const ok = validate(obj);
  if (ok) return { valid: true };
  const msg = (validate.errors || []).map(e => `${e.instancePath} ${e.message}`).join("; ");
  return { valid: false, errors: msg };
}

import { Recipe } from "./retriever";

export function composePrompt(userPrompt: string, examples: Recipe[]): string {
  const fewShots = examples.map((ex, idx) => `Example ${idx + 1} NL:\n${ex.prompt}\nExample ${idx + 1} JSON:\n${JSON.stringify(ex.workflow, null, 2)}`).join("\n\n");
  return [
    "You convert natural language requests into a Workflow JSON that matches the schema.",
    "Return ONLY valid JSON. No markdown. No explanations.",
    fewShots,
    `User NL:\n${userPrompt}\nJSON:`
  ].filter(Boolean).join("\n\n");
}

import { Retriever } from "./retriever";
import { composePrompt } from "./prompt";
import { callLLM } from "./llm";
import { validateWorkflow } from "./validation";

function extractJson(text: string): string {
  // Strip fences if model added them.
  const m = text.match(/\{[\s\S]*\}/);
  return m ? m[0] : text.trim();
}

export async function generateWorkflow(userPrompt: string, opts?: { k?: number; model?: string; debug?: boolean }) {
  const retriever = new Retriever();
  const k = opts?.k ?? 3;
  const examples = retriever.retrieve(userPrompt, k);
  const prompt = composePrompt(userPrompt, examples);
  const res = await callLLM(prompt, opts?.model);
  const raw = extractJson(res.raw);

  let obj: any;
  try { obj = JSON.parse(raw); } catch (e) {
    if (opts?.debug) console.error("Parse error; attempting compact repair");
    // Minimal repair: remove trailing junk then retry parse
    const compact = raw.replace(/```json|```/g, "").trim();
    obj = JSON.parse(compact);
  }

  const v1 = validateWorkflow(obj);
  if (v1.valid) return { workflow: obj, examples };

  // Open-ended repair path: provide errors back to model once.
  const repairMsg = `The previous JSON failed validation: ${v1.errors}. Please return a corrected JSON only.`;
  const repair = await callLLM(composePrompt(userPrompt + "\n" + repairMsg, examples), opts?.model);
  const raw2 = extractJson(repair.raw);
  const obj2 = JSON.parse(raw2);
  const v2 = validateWorkflow(obj2);
  return { workflow: obj2, examples, valid: v2.valid, errors: v2.errors };
}

src/eval.ts (simple scoring; candidate can enhance)

import fs from "fs";
import path from "path";
import { generateWorkflow } from "./generateWorkflow";

interface Gold { trigger: any; actions: any[] }

function normStr(s: any) { return String(s ?? "").trim().toLowerCase(); }
function setOverlap(a: string[] = [], b: string[] = []) {
  const A = new Set(a.map(normStr));
  const B = new Set(b.map(normStr));
  let inter = 0; A.forEach(x => { if (B.has(x)) inter++; });
  return inter / Math.max(1, Math.max(A.size, B.size));
}

function scorePair(pred: any, gold: Gold) {
  const out = { schemaValid: true, triggerType: 0, actionTypes: 0, fields: 0 };
  // types
  out.triggerType = pred?.trigger?.type === gold.trigger.type ? 1 : 0;
  const predActs = (pred?.actions || []).map((a: any) => a.type);
  const goldActs = (gold.actions || []).map(a => a.type);
  out.actionTypes = setOverlap(predActs, goldActs);
  // field-level: naive check of common fields
  const pcond = pred?.trigger?.conditions?.[0];
  const gcond = gold?.trigger?.conditions?.[0];
  let fieldPts = 0, total = 0;
  if (gcond?.field) { total++; if (normStr(pcond?.field) === normStr(gcond.field)) fieldPts++; }
  if (gcond?.operator) { total++; if (normStr(pcond?.operator) === normStr(gcond.operator)) fieldPts++; }
  if (gcond?.value) { total++; const pv = Array.isArray(pcond?.value) ? pcond.value : [pcond?.value]; const gv = Array.isArray(gcond?.value) ? gcond.value : [gcond?.value]; if (setOverlap(pv.map(String), gv.map(String)) > 0.66) fieldPts++; }
  out.fields = total ? fieldPts / total : 1;
  return out;
}

async function main() {
  const prompts: string[] = JSON.parse(fs.readFileSync(path.resolve("data/eval_prompts.json"), "utf-8"));
  const gold: Gold[] = JSON.parse(fs.readFileSync(path.resolve("data/eval_gold.json"), "utf-8"));

  let validCount = 0; let sumTrig = 0; let sumActs = 0; let sumFields = 0;

  for (let i = 0; i < prompts.length; i++) {
    const p = prompts[i];
    process.stdout.write(`\n[${i+1}/${prompts.length}] ${p}\n`);
    try {
      const { workflow } = await generateWorkflow(p, { k: 3 });
      const s = scorePair(workflow, gold[i]);
      const schemaValid = true; // generateWorkflow already validated internally
      validCount += schemaValid ? 1 : 0;
      sumTrig += s.triggerType; sumActs += s.actionTypes; sumFields += s.fields;
      console.log(`→ triggerType=${s.triggerType.toFixed(2)} actionTypes=${s.actionTypes.toFixed(2)} fields=${s.fields.toFixed(2)}`);
    } catch (e:any) {
      console.log(`✖ error: ${e.message}`);
    }
  }

  const n = prompts.length;
  console.log("\n==== SUMMARY ====");
  console.log(`Schema Validity: ${(validCount/n*100).toFixed(0)}%`);
  console.log(`Trigger Type Acc: ${(sumTrig/n*100).toFixed(0)}%`);
  console.log(`Action Types Overlap: ${(sumActs/n*100).toFixed(0)}%`);
  console.log(`Field Match: ${(sumFields/n*100).toFixed(0)}%`);
}

main();
